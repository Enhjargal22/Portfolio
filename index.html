@Service
class FinancialEngine(
private val ledgerRepo: TransactionRepository,
private val rateService: CurrencyExchangeService
) {
suspend fun rebalancePortfolio(userId: Long): FinancialSummary {
val rates = rateService.getActiveRates()
val balance = ledgerRepo.calculateAvailableBalance(userId, rates)
return FinancialSummary(balance, status = LedgerStatus.HEALTHY)
}
}
